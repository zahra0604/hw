def sum_numbers(n):

    total = 0

    i = 1

    while i <= n:

        if i % 3 == 0:

            total += i

        i += 1

    return total

def factorial(n):

    result = 1

    i = 1

    while i <= n:

        result *= i

        i += 1

    return result

number = int(input("Enter a number: "))

print(factorial(number))
